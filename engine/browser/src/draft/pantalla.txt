/* ══════════════════════════════════════════════════════════════════
   MODO DRAFT · RETRO PACK 2 — la pantalla

   Las reglas, la IA y el evento viven en src/draft/ (se prueban en node
   con check-draft.mjs). Aquí solo se pinta, se guarda y se lanza el duelo
   de siempre con `alTerminar`, igual que el torneo suizo.

   Fases (en `goat.draft.v1`):
     inicio → draft (54 picks) → construir → evento (suizo, Bo3, reloj) → fin
   ══════════════════════════════════════════════════════════════════ */
var DRAFT_ST;              // el estado guardado (perezoso: ver la trampa de pintarMenu)
var DRAFT_D;               // catálogo + análisis del formato (se calcula una vez)
var DRAFT_OCUPADO = false; // los bots están eligiendo / jugando
var DRAFT_SEL = null;      // índice de la carta marcada en el sobre
var DRAFT_SUG = null;      // sugerencia del modo entrenamiento para este paso
var DRAFT_VER = null;      // carta del panel de detalle
var DRAFT_CLAVE = "goat.draft.v1";
const DRAFT_NOMBRES = ["Akira","Bea","Carlos","Dana","Elio","Fátima","Gonzalo","Hana","Íñigo","Jade","Kenji",
  "Lucía","Mateo","Nora","Óscar","Paula","Quim","Rocío","Saúl","Tere","Unai","Vera","Wen","Ximena"];

function draftD(){ return DRAFT_D ??= Draft.prepararDraft(RP02); }
function draftCargar(){
  if(DRAFT_ST === undefined){
    try{ DRAFT_ST = JSON.parse(localStorage.getItem(DRAFT_CLAVE) || "null"); }catch(e){ DRAFT_ST = null; }
    if(DRAFT_ST && !Draft.sanearMesa(DRAFT_ST.mesa)) DRAFT_ST = null;
  }
  return DRAFT_ST;
}
function draftGuardar(){
  try{
    if(DRAFT_ST) localStorage.setItem(DRAFT_CLAVE, JSON.stringify(DRAFT_ST));
    else localStorage.removeItem(DRAFT_CLAVE);
  }catch(e){
    /* el log de picks es lo que más ocupa: si no cabe, se recorta y se reintenta */
    try{ DRAFT_ST.mesa.log = DRAFT_ST.mesa.log.map(l => ({ asiento:l.asiento, ronda:l.ronda, paso:l.paso, uid:l.uid, id:l.id, motivo:l.motivo }));
         localStorage.setItem(DRAFT_CLAVE, JSON.stringify(DRAFT_ST)); }catch(e2){}
  }
}
const dEsc = x => String(x ?? "").replace(/[&<>"]/g, c => ({ "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;" })[c]);
const dCarta = id => draftD().cat.get(id);
const dNombre = id => { const c = dCarta(id); return c ? (c.es || c.nombre) : (NAMES[id]?.name ?? id); };
const dImg = id => globalThis.imagenDeCarta ? globalThis.imagenDeCarta(dCarta(id)?.oficial ?? id, "cards_small") : "";
function dNombreAsiento(a){
  const st = DRAFT_ST; if(!st) return "";
  if(a === st.mesa.cfg.humano) return st.miNombre || "Tú";
  return DRAFT_NOMBRES[(a + Number(String(Draft.hashTexto?.(st.mesa.semilla) ?? 0).slice(-2))) % DRAFT_NOMBRES.length] + " · bot " + a;
}

/* ── EMPEZAR ── */
function nuevoDraft(op){
  const semilla = (op.semilla || "").trim() || ("RP02-" + Date.now().toString(36).toUpperCase());
  const mesa = Draft.nuevaMesa(RP02, { semilla, asientos: op.asientos, humano: 0, mainMin: op.mainMin,
    rondasEvento: op.rondas, minutosRonda: op.minutos, turnosExtra: op.turnosExtra, entrenamiento: op.entrenamiento });
  DRAFT_ST = { v:1, fase:"draft", mesa, miNombre: (op.nombre || "").trim().slice(0, 18) || "Tú",
               mazos:null, miMazo:null, evento:null, match:null, partidasLog:[] };
  DRAFT_SEL = null; DRAFT_SUG = null;
  draftGuardar(); pintarDraft();
}
function abandonarDraft(){
  if(!confirm("¿Borrar este draft? Se pierde el pool, el mazo y el evento.")) return;
  DRAFT_ST = null; draftGuardar(); pintarDraft();
}

/* ── UN PICK ──
   Los bots eligen con el estado de ANTES del pase (aplicarPaso lo hace
   todo a la vez). Se calcula de uno en uno cediendo al navegador entre
   bot y bot para que la pantalla no se congele. */
async function elegirDraft(indice){
  const st = DRAFT_ST; if(!st || DRAFT_OCUPADO || st.fase !== "draft") return;
  const mesa = st.mesa, D = draftD(), humano = mesa.cfg.humano;
  if(!(indice >= 0 && indice < mesa.sobres[humano].length)) return;
  DRAFT_OCUPADO = true; pintarDraft();
  const elecciones = { [humano]: indice }, registros = { [humano]: { motivo: DRAFT_SUG ? `(tu pick) · la IA sugería: ${DRAFT_SUG.explicacion}` : "(tu pick)" } };
  try{
    for(let a = 0; a < mesa.cfg.asientos; a++){
      if(a === humano) continue;
      const r = Draft.picksDeBots(mesa, D, { saltar: new Set([...Array(mesa.cfg.asientos).keys()].filter(x => x !== a)) });
      elecciones[a] = r.elecciones[a]; registros[a] = r.registros[a];
      const p = document.getElementById("dProgreso");
      if(p) p.textContent = `Los demás eligen… ${a}/${mesa.cfg.asientos - 1}`;
      await new Promise(res => setTimeout(res, 0));
    }
    Draft.aplicarPaso(mesa, RP02, elecciones, registros);
  }finally{ DRAFT_OCUPADO = false; }
  DRAFT_SEL = null; DRAFT_SUG = null;
  if(mesa.terminado) terminarDraft();
  draftGuardar(); pintarDraft();
}
/* Al acabar: cada bot construye su mazo (y te sugiere el tuyo). */
function terminarDraft(){
  const st = DRAFT_ST, D = draftD();
  st.mazos = {};
  for(let a = 0; a < st.mesa.cfg.asientos; a++) st.mazos[a] = Draft.construirAsiento(st.mesa, D, a);
  const yo = st.mazos[st.mesa.cfg.humano];
  st.miMazo = { main:[...yo.main], side:[...yo.side] };
  st.fase = "construir";
}
/* Ver lo que piensa la IA de tu sobre (modo entrenamiento). */
async function sugerirDraft(){
  const st = DRAFT_ST; if(!st || st.fase !== "draft" || DRAFT_SUG) return;
  await new Promise(res => setTimeout(res, 0));
  DRAFT_SUG = Draft.sugerirPick(st.mesa, draftD(), st.mesa.cfg.humano);
  pintarDraft();
}

  pintarDraftSinTraducir();

function pintarDraftSinTraducir(){
  const caja = document.getElementById("dCaja"); if(!caja) return;
  const st = draftCargar();
  if(!st) return pintarInicioDraft(caja);
  if(st.fase === "draft") return pintarPicks(caja, st);
  if(st.fase === "construir" || st.fase === "side") return pintarConstruir(caja, st);
  if(st.fase === "evento" || st.fase === "fin") return pintarEvento(caja, st);
}

/* La etiqueta de dependencias de una carta para TU pool (información
   pública: las reglas y tus cartas). */
function etiquetaDep(id, poolIds){
  const D = draftD(), c = dCarta(id); if(!c) return "";
  const k = Draft.categoriaDe(c, poolIds.map(dCarta), { formato: D.formato }).k;
  return { 1:'<span class="dDep d1" title="no hay ruta de invocación en el pool de RP02">sin ruta</span>',
           2:'<span class="dDep d2" title="le faltan piezas que no tienes">falta pieza</span>',
           3:'<span class="dDep d3" title="tienes las piezas, pero rara vez llegan juntas">poco consistente</span>',
           4:"" }[k] ?? "";
}
function tarjeta(id, { extra = "", clase = "", attrs = "" } = {}){
  const c = dCarta(id);
  return `<div class="dCarta ${clase}" data-id="${id}" ${attrs}>
    <img loading="lazy" src="${dImg(id)}" alt="" onerror="this.style.visibility='hidden'">
    <span class="dCn">${dEsc(dNombre(id))}</span>
    <span class="dCr r${c?.rareza}">${c?.rareza ?? ""}</span>${extra}</div>`;
}
function pintarDetalle(){
  const z = document.getElementById("dDetalle"); if(!z) return;
  const id = DRAFT_VER; const c = id && dCarta(id);
  if(!c){ z.innerHTML = '<p class="dNota">Pasa el ratón (o toca) una carta para leerla.</p>'; return; }
  const stats = c.esMonstruo ? `Nivel ${c.nivel} · ${c.atk}/${c.def}${c.tributos ? ` · ${c.tributos} tributo(s)` : c.tributos === null ? " · invocación especial" : ""}` :
    c.esMagia ? "Magia" : "Trampa";
  const f = draftD().formato.get(c.nombre);
  z.innerHTML = `<img src="${globalThis.imagenDeCarta ? globalThis.imagenDeCarta(c.oficial, "cards") : ""}" alt="">
    <b>${dEsc(c.es)}</b><small>${dEsc(c.nombre)} · ${c.rareza}</small>
    <small>${dEsc(stats)}</small>
    <p>${dEsc(NAMES[c.id]?.desc ?? c.texto)}</p>
    ${f && !f.ok ? `<p class="dAviso">Sin ruta en este pool: ${dEsc(f.motivo)}</p>` : ""}
    <p class="dNota">${dEsc(c.nota ?? "")}</p>`;
}
function cablearCartas(raiz, alPulsar){
  for(const el of raiz.querySelectorAll(".dCarta")){
    const id = Number(el.dataset.id);
    el.onmouseenter = () => { DRAFT_VER = id; pintarDetalle(); };
    el.onclick = ev => { DRAFT_VER = id; pintarDetalle(); alPulsar?.(el, id, ev); };
  }
}
function poolAgrupado(ids, { conDep = true } = {}){
  const grupos = [["Monstruos sin tributo", c => c.esMonstruo && c.tributos === 0],
                  ["Monstruos de tributo", c => c.esMonstruo && c.tributos > 0],
                  ["Invocación especial / ritual / fusión", c => c.esMonstruo && c.tributos === null],
                  ["Magias", c => c.esMagia], ["Trampas", c => c.esTrampa]];
  const cuenta = new Map(); for(const id of ids) cuenta.set(id, (cuenta.get(id) ?? 0) + 1);
  return grupos.map(([t, f]) => {
    const dentro = [...cuenta].filter(([id]) => f(dCarta(id))).sort((a, b) => dCarta(b[0]).base - dCarta(a[0]).base);
    if(!dentro.length) return "";
    return `<div class="dGrupo"><h4>${t} (${dentro.reduce((s, x) => s + x[1], 0)})</h4>` +
      dentro.map(([id, n]) => `<div class="dFilaPool dCarta" data-id="${id}"><span>${n > 1 ? n + "× " : ""}${dEsc(dNombre(id))}</span>${conDep ? etiquetaDep(id, ids) : ""}</div>`).join("") + "</div>";
  }).join("");
}

/* ── TUS PICKS, SOBRE A SOBRE ──
   E: «solo falta ver mis cartas que ya he pickeado en cada ronda». Cada
   sobre con sus cartas en el orden en que las cogiste (el número es el
   pick). La vista elegida se recuerda en este navegador. */
function vistaPool(){ try{ return localStorage.getItem("goat.draft.vistaPool") || "sobre"; }catch(e){ return "sobre"; } }
function ponerVistaPool(v){ try{ localStorage.setItem("goat.draft.vistaPool", v); }catch(e){} }
function poolPorSobre(m, h, entreno){
  const pool = m.picks[h].map(uid => m.instancias[uid]);
  const porSobre = Array.from({ length: m.cfg.sobres }, () => []);
  for(const l of m.log) if(l.asiento === h) porSobre[l.ronda]?.push(l);
  const hasta = m.terminado ? m.cfg.sobres - 1 : m.ronda;
  let html = "";
  for(let r = hasta; r >= 0; r--){
    const picks = porSobre[r];
    const dir = Draft.direccion(r) === "izquierda" ? "→" : "←";
    html += `<div class="dRonda"><h4>Sobre ${r + 1} ${dir} · ${picks.length}/${m.cfg.cartasPorSobre}${r === m.ronda && !m.terminado ? " (en curso)" : ""}</h4>` +
      (picks.length ? `<div class="dSobre peq">${picks.map(l => tarjeta(l.id, {
        extra: `<span class="dVal" title="pick ${l.paso + 1}">${l.paso + 1}</span>` + (entreno ? etiquetaDep(l.id, pool) : "") })).join("")}</div>`
        : `<p class="dNota">Aún no has cogido nada de este sobre.</p>`) + `</div>`;
  }
  return `<div class="dPorSobre">${html}</div>`;
}

function pintarPicks(caja, st){

  const pool = m.picks[h].map(uid => m.instancias[uid]);
  const dir = Draft.direccion(m.ronda) === "izquierda" ? "pasa a la izquierda →" : "← pasa a la derecha";
  const entreno = m.cfg.entrenamiento;
  caja.innerHTML = `
    <div class="dCab">
      <div><b>Sobre ${m.ronda + 1}/${m.cfg.sobres}</b> · pick ${m.paso + 1}/${m.cfg.cartasPorSobre} · ${dir}
        <span class="dNota"> · ${m.cfg.asientos} jugadores · semilla ${dEsc(m.semilla)}</span></div>
      <div class="dBotonesCab">
        ${entreno ? `<button class="dBtn" id="dSugerir" ${DRAFT_SUG ? "disabled" : ""}>¿Qué cogería la IA?</button>` : ""}
        <button class="dBtn" id="dSalir">Menú</button>
      </div>
    </div>

      <div class="dIzq">
        <div class="dSobre" id="dSobre">${sobre.map((id, i) => tarjeta(id, { clase: i === DRAFT_SEL ? "sel" : "",
            extra: entreno ? etiquetaDep(id, pool) : "", attrs: `data-i="${i}"` })).join("")}</div>
        <div class="dAcciones">
          <button class="mbig" id="dElegir" ${DRAFT_SEL === null || DRAFT_OCUPADO ? "disabled" : ""}>${DRAFT_OCUPADO ? "…" : "Elegir esta carta"}</button>
          <span id="dProgreso" class="dNota">${DRAFT_OCUPADO ? "Los demás eligen…" : "Toca una carta y pulsa elegir (o doble clic)."}</span>
        </div>
        ${DRAFT_SUG ? `<div class="dSug"><b>La IA cogería ${dEsc(dNombre(DRAFT_SUG.id))}.</b> ${dEsc(DRAFT_SUG.explicacion)}
             <small>${DRAFT_SUG.planes.map(p => `${Draft.PLANES[p.plan].corto} ${Math.round(p.peso * 100)} %`).join(" · ")}</small></div>` : ""}
        <div class="dPoolCab"><h3>Tu pool (${pool.length}/${m.cfg.sobres * m.cfg.cartasPorSobre})</h3>

  });
  for(const z of caja.querySelectorAll(".dPool, .dPorSobre")) cablearCartas(z);

  pintarDetalle();
  cablearCartas(document.getElementById("dMain"), el => { const i = Number(el.dataset.i); side.push(main.splice(i, 1)[0]); draftGuardar(); pintarDraft(); });
  cablearCartas(document.getElementById("dSide"), el => { const i = Number(el.dataset.i); main.push(side.splice(i, 1)[0]); draftGuardar(); pintarDraft(); });
  document.getElementById("dSugerido").onclick = () => { st.miMazo = { main:[...sug.main], side:[...sug.side] }; draftGuardar(); pintarDraft(); };
  document.getElementById("dLog").onclick = () => descargarLogDraft();
  document.getElementById("dSalir").onclick = cerrarDraft;
  document.getElementById("dListo").onclick = () => {
    if(!legal) return;
    if(sideando){ st.fase = "evento"; draftGuardar(); return jugarPartidaDraft(); }
    if(!st.evento) empezarEventoDraft();
    st.registro = { main:[...main], side:[...side] };      // el mazo con el que empieza cada match
    st.fase = "evento"; draftGuardar(); pintarDraft();
  };
}

/* ── EL LOG ──
   Los picks de todos, con el porqué de cada bot. Solo cuando el draft ha
   terminado: durante el draft no se filtra nada de nadie. */
function descargarLogDraft(){
  const st = DRAFT_ST; if(!st?.mesa?.terminado) return;
  const D = draftD(), m = st.mesa;
  const L = [`DRAFT RETRO PACK 2 · semilla ${m.semilla} · ${m.cfg.asientos} jugadores`, `modelo de sobres: ${m.cfg.modelo.nombre}`, ""];
  for(let a = 0; a < m.cfg.asientos; a++){
    const mz = st.mazos?.[a];
    L.push(`══ ${dNombreAsiento(a)} ══ ${mz ? `${mz.nombrePlan} (nota ${mz.total.toFixed(0)})` : ""}`);
    for(const l of m.log.filter(x => x.asiento === a))
      L.push(`  S${l.ronda + 1}P${l.paso + 1}  ${dNombre(l.id)}  — ${l.motivo ?? ""}` +
             (l.candidatos ? `\n          candidatas: ${[].concat(l.candidatos).join(" | ")}\n          planes: ${l.planes ?? ""}` : ""));
    if(mz) L.push("  MAIN: " + mz.main.map(dNombre).join(", "), "  SIDE: " + mz.side.map(dNombre).join(", "));
    L.push("");
  }
  bajarTexto(`draft-rp02-${m.semilla}.txt`, L.join("\n"));
}

/* ── EL EVENTO ── */
function empezarEventoDraft(){
  const st = DRAFT_ST, m = st.mesa;
  const jugadores = [...Array(m.cfg.asientos).keys()].map(a => ({ id:a, nombre: dNombreAsiento(a), yo: a === m.cfg.humano }));
  st.evento = Draft.nuevoEvento({ jugadores, rondas: m.cfg.rondasEvento, semilla: m.semilla + "|evento" });
  Draft.emparejar(st.evento);
  st.match = null;
}
function miMesaDraft(){ const st = DRAFT_ST; return st?.evento ? Draft.mesaDe(st.evento, st.mesa.cfg.humano) : null; }
function rivalDeMesa(mesa){ const h = DRAFT_ST.mesa.cfg.humano; return mesa.a === h ? mesa.b : mesa.a; }
function marcadorMatch(mt){
  const g = mt.juegos.filter(j => j.gane && !j.empate).length, p = mt.juegos.filter(j => !j.gane && !j.empate).length;
  return { g, p, e: mt.juegos.filter(j => j.empate).length };
}
function matchDecidido(mt){
  const { g, p, e } = marcadorMatch(mt);
  if(g >= 2 || p >= 2) return true;
  if(mt.fin && Date.now() >= mt.fin) return true;          // se acabó el tiempo de la ronda
  return mt.juegos.length >= 5 || e >= 2;
}

function jugarPartidaDraft(){
  const st = DRAFT_ST, mesa = miMesaDraft(); if(!mesa || mesa.bye !== undefined) return;
  const rival = rivalDeMesa(mesa), mzR = st.mazos[rival];
  st.match ??= { ronda: st.evento.ronda, mesa: mesa.mesa, rival, juegos: [], fin: null,
                 empiezo: ((Draft.hashTexto?.(st.mesa.semilla + st.evento.ronda) ?? 0) % 2) === 0 };
  const mt = st.match;
  if(!mt.fin) mt.fin = Date.now() + st.mesa.cfg.minutosRonda * 60000;
  draftGuardar();
  const config = { ...cfg, nivel:"experto", modo:"draft", nombreMazo:"Draft RP02", nombreRival: dNombreAsiento(rival),
    planRival: mzR.planDuelo, empiezasTu: mt.empiezo, lpIniciales: st.mesa.cfg.lp, lpRival: st.mesa.cfg.lp,
    relojRonda: { fin: mt.fin, turnosExtra: st.mesa.cfg.turnosExtra },
    alTerminar: (ganaste, datos = {}) => terminarPartidaDraft(ganaste, datos),
    avatarMio:{ src:avatarSrc(cfg.avatar), nombre:avatarNombre(cfg.avatar) },
    avatarRival:{ src:avatarSrc(AVATAR_IA), nombre: dNombreAsiento(rival) } };
  if(typeof globalThis.__LANZAR_DRAFT__ === "function"){
    globalThis.__LANZAR_DRAFT__({ deck: st.miMazo.main, deckRival: mzR.main, config }); return;
  }
  document.getElementById("draft").hidden = true;
  document.getElementById("menu").style.display = "none";
  if(globalThis.matchMedia?.("(pointer:coarse)")?.matches || (globalThis.innerWidth??1920)<900) asegurarPantallaCompleta();
  boot({ createCore: __OCG__.createCore, Xns: __OCG__, GoatDuel, makeAutoPlayer, makeTrivialResolver, scriptReader,
         cardsRaw: CARDS, names: NAMES, deck: [...st.miMazo.main], extra: [], deckRival: [...mzR.main], extraRival: [],
         View, config })
  .catch(e => { document.getElementById("boot").innerHTML =
    '<div style="color:#ff6a55;font-size:13px;max-width:640px;text-align:left"><b>Error al arrancar</b><br><pre style="white-space:pre-wrap">'+e.stack+'</pre></div>'; });
}
function terminarPartidaDraft(ganaste, datos){
  const st = DRAFT_ST, mt = st?.match;
  View.cerrarFinal?.();
  document.getElementById("draft").hidden = false;
  if(mt){
    mt.juegos.push({ gane: !!ganaste && !datos.empate, empate: !!datos.empate, motivo: datos.motivo ?? "",
                     porTiempo: !!datos.porTiempo, turnos: datos.turnos ?? null, lpMio: datos.lpMio ?? null, lpRival: datos.lpRival ?? null });
    if(!datos.empate) mt.empiezo = !ganaste;            // empieza el que perdió
    if(matchDecidido(mt)) cerrarMiMatch();
    else st.fase = "side";                              // side entre partidas
  }
  draftGuardar(); pintarDraft();
}
function cerrarMiMatch(){
  const st = DRAFT_ST, mt = st.match, h = st.mesa.cfg.humano;
  const mesa = miMesaDraft();
  const { g, p, e } = marcadorMatch(mt);
  const yoA = mesa.a === h;
  Draft.apuntar(st.evento, mesa.mesa, { ga: yoA ? g : p, gb: yoA ? p : g, empates: e,
    motivo: mt.fin && Date.now() >= mt.fin && g === p ? "tiempo" : "" });
  st.ultimoMatch = { ...mt, resultado: g > p ? "victoria" : g < p ? "derrota" : "empate" };
  st.match = null;
  /* el side se deshace al acabar el match: cada match empieza con el
     mazo registrado */
  if(st.registro) st.miMazo = { main:[...st.registro.main], side:[...st.registro.side] };
  st.fase = "evento";
}

/* Las demás mesas: partidas de verdad entre bots, con su plan. */
var DRAFT_DEPS;
async function depsDraft(){
  if(DRAFT_DEPS) return DRAFT_DEPS;
  const lib = await __OCG__.createCore({ sync:true });
  const db = new Map(); for(const k in CARDS){ const c = CARDS[k]; db.set(c.code, { ...c, race: BigInt(c.race) }); }
  return DRAFT_DEPS = { lib, X: __OCG__, GoatDuel, crearCerebro, trivial: makeTrivialResolver(__OCG__),
                        generico: makeAutoPlayer(__OCG__), db, names: NAMES, scriptReader };
}
async function jugarOtrasMesas(){
  const st = DRAFT_ST, ev = st.evento; if(DRAFT_OCUPADO) return;
  DRAFT_OCUPADO = true; pintarDraft();
  try{
    const deps = await depsDraft();
    const pendientes = ev.mesas.filter(m => !m.resultado);
    let n = 0;
    for(const m of pendientes){
      const A = st.mazos[m.a], B = st.mazos[m.b];
      const r = await Draft.matchDeBots(deps, { main:A.main, plan:A.planDuelo }, { main:B.main, plan:B.planDuelo },
        { semilla: (Draft.hashTexto?.(`${st.mesa.semilla}|${ev.ronda}|${m.mesa}`) ?? (ev.ronda * 97 + m.mesa)) % 2147483647 || 1,
          bo: st.mesa.cfg.bo, maxTurnos: 50, ceder: () => new Promise(res => setTimeout(res, 0)) });
      Draft.apuntar(ev, m.mesa, { ga: r.ga, gb: r.gb, empates: r.empates, motivo: r.cortes ? `${r.cortes} corte(s) técnico(s) por LP` : "" });
      n++;
      const p = document.getElementById("dProgreso"); if(p) p.textContent = `Mesas jugadas: ${n}/${pendientes.length}`;
      await new Promise(res => setTimeout(res, 0));
    }
  }finally{ DRAFT_OCUPADO = false; }
  draftGuardar(); pintarDraft();
}
function siguienteRondaDraft(){
  const st = DRAFT_ST, ev = st.evento;
  if(!Draft.rondaCompleta(ev)) return;
  Draft.cerrarRonda(ev);
  st.ultimoMatch = null;
  if(ev.fase === "fin") st.fase = "fin";
  else Draft.emparejar(ev);
  draftGuardar(); pintarDraft();
}

function pintarEvento(caja, st){
  const ev = st.evento, h = st.mesa.cfg.humano;
  const cl = Draft.clasificacion(ev);
  const mesa = ev.fase === "fin" ? null : miMesaDraft();
  const pct = x => (100 * x).toFixed(1) + " %";
  let miParte = "";
  if(ev.fase === "fin"){
    const pos = cl.findIndex(f => f.yo) + 1;
    miParte = `<div class="dSug"><b>Fin del evento: quedas ${pos}.º de ${cl.length}.</b>
      Campeón: ${dEsc(cl[0].nombre)} con ${dEsc(st.mazos[cl[0].id].nombrePlan)}.</div>`;
  } else if(mesa){
    const yaMio = mesa.bye !== undefined || mesa.resultado;
    if(mesa.bye !== undefined) miParte = `<div class="dSug"><b>Ronda ${ev.ronda + 1}: bye</b> (cuenta como 2-0).</div>`;
    else {
      const rival = rivalDeMesa(mesa), mt = st.match;
      const marc = mt ? marcadorMatch(mt) : null;
      const reloj = mt?.fin ? Math.max(0, Math.round((mt.fin - Date.now()) / 60000)) : null;
      miParte = `<div class="dSug"><b>Ronda ${ev.ronda + 1}/${ev.rondas} · mesa ${mesa.mesa}: contra ${dEsc(dNombreAsiento(rival))}</b>
        ${mesa.resultado ? ` — ${st.ultimoMatch?.resultado ?? ""} ${mesa.a === h ? mesa.resultado.ga : mesa.resultado.gb}-${mesa.a === h ? mesa.resultado.gb : mesa.resultado.ga}`
          : marc ? ` — vas ${marc.g}-${marc.p}${marc.e ? ` (${marc.e} empate)` : ""}${reloj !== null ? ` · quedan ${reloj} min de ronda` : ""}` : ""}
        ${!mesa.resultado ? `<div class="dAcciones"><button class="mbig" id="dJugar">${mt ? `Jugar la partida ${mt.juegos.length + 1}` : "Jugar el match (Bo3)"}</button>
          ${mt ? `<button class="dBtn" id="dSide">Cambiar el side</button>` : `<button class="dBtn" id="dEditar">Revisar el mazo</button>`}</div>` : ""}</div>`;
    }
    if(yaMio){
      const quedan = ev.mesas.filter(m => !m.resultado).length;
      miParte += `<div class="dAcciones">${quedan ? `<button class="mbig" id="dOtras" ${DRAFT_OCUPADO ? "disabled" : ""}>Jugar las otras ${quedan} mesas</button>`
        : `<button class="mbig" id="dSiguiente">${ev.ronda + 1 >= ev.rondas ? "Ver la clasificación final" : "Siguiente ronda"}</button>`}
        <span id="dProgreso" class="dNota">${DRAFT_OCUPADO ? "Jugando…" : quedan ? "Se juegan de verdad, bot contra bot, con el motor." : ""}</span></div>`;
    }
  }
  const mesas = (ev.fase === "fin" ? (ev.historial.at(-1) ?? []) : ev.mesas).map(m => m.bye !== undefined
    ? `<tr><td>${m.mesa}</td><td colspan="3">${dEsc(dNombreAsiento(m.bye))} — bye</td></tr>`
    : `<tr${m.a === h || m.b === h ? ' class="yo"' : ""}><td>${m.mesa}</td><td>${dEsc(dNombreAsiento(m.a))}</td>
        <td>${m.resultado ? `${m.resultado.ga}-${m.resultado.gb}${m.resultado.empates ? "-" + m.resultado.empates : ""}` : "·"}</td>
        <td>${dEsc(dNombreAsiento(m.b))}${m.resultado?.motivo ? ` <small>(${dEsc(m.resultado.motivo)})</small>` : ""}</td></tr>`).join("");
  caja.innerHTML = `
    <div class="dCab"><div><b>Evento · suizo de ${ev.rondas} rondas · Bo${st.mesa.cfg.bo}</b> · ${ev.jugadores.length} jugadores</div>
      <div class="dBotonesCab"><button class="dBtn" id="dLog">Log del draft</button>
        ${ev.fase === "fin" ? "" : `<button class="dBtn" id="dRetirar">Retirarme</button>`}
        <button class="dBtn" id="dNuevo">Nuevo draft</button><button class="dBtn" id="dSalir">Menú</button></div></div>
    ${miParte}
    <div class="dDos">
      <div><h3>Mesas ${ev.fase === "fin" ? "de la última ronda" : `de la ronda ${ev.ronda + 1}`}</h3>
        <table class="dTabla"><tr><th>#</th><th>Jugador</th><th></th><th>Jugador</th></tr>${mesas}</table></div>
      <div><h3>Clasificación</h3>
        <table class="dTabla"><tr><th></th><th>Jugador</th><th>Pts</th><th>V-D-E</th><th title="% de victorias de tus rivales (suelo 33 %)">OMW</th><th title="lo mismo de los rivales de tus rivales">OOMW</th><th title="% de partidas ganadas">GW</th>${ev.fase === "fin" ? "<th>Mazo</th>" : ""}</tr>
        ${cl.map((f, i) => `<tr${f.yo ? ' class="yo"' : ""}><td>${i + 1}</td><td>${dEsc(f.nombre)}${f.retirado ? " (ret.)" : ""}</td><td>${f.puntos}</td>
          <td>${f.ganados}-${f.perdidos}-${f.empatados}</td><td>${pct(f.omw)}</td><td>${pct(f.oomw)}</td><td>${pct(f.gw)}</td>
          ${ev.fase === "fin" ? `<td><small>${dEsc(Draft.PLANES[st.mazos[f.id].plan].corto)}</small></td>` : ""}</tr>`).join("")}</table>
        <p class="dNota">Desempates: puntos (victoria 3, empate 1) → OMW% → OOMW% → GW%. Bye = victoria 2-0, fuera de los porcentajes.
          Las mesas de bots se juegan con el motor; un duelo que llega a 50 turnos es un corte técnico y se decide por LP (no es un deck-out).</p>
      </div>
    </div>`;
  const on = (id, f) => { const b = document.getElementById(id); if(b) b.onclick = f; };
  on("dJugar", jugarPartidaDraft);
  on("dSide", () => { st.fase = "side"; draftGuardar(); pintarDraft(); });
  on("dEditar", () => { st.fase = "construir"; draftGuardar(); pintarDraft(); });
  on("dOtras", jugarOtrasMesas);
  on("dSiguiente", siguienteRondaDraft);
  on("dLog", descargarLogDraft);
  on("dSalir", cerrarDraft);
  on("dNuevo", abandonarDraft);
  on("dRetirar", () => {
    if(!confirm("¿Retirarte? Tus rondas restantes no se juegan y el evento sigue sin ti.")) return;
    if(st.match){ st.match = null; }
    const m = miMesaDraft();
    if(m && !m.resultado && m.bye === undefined){ const yoA = m.a === h; Draft.apuntar(ev, m.mesa, { ga: yoA ? 0 : 2, gb: yoA ? 2 : 0, motivo:"retirada" }); }
    Draft.retirar(ev, h); draftGuardar(); pintarDraft();
  });
}
